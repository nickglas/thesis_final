% --- Line breaking for long code tokens (paths, identifiers, hashes) ---
% Long snake_case identifiers and file paths inside \texttt{} are otherwise
% unbreakable and overflow the margin. Scope a breakable underscore to
% typewriter text only (leaving \_ in math mode, e.g. \mathrm{round\_index},
% untouched) so such tokens wrap without an inserted hyphen.
\makeatletter
\NewCommandCopy{\thesis@orig@texttt}{\texttt}
\DeclareRobustCommand{\texttt}[1]{%
  \begingroup
    \def\_{\textunderscore\penalty\z@\hskip\z@skip}%
    \thesis@orig@texttt{#1}%
  \endgroup
}
\makeatother
